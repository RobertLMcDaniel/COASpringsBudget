        path.setAttribute("tabindex", "0");
        path.setAttribute("role", "button");
        path.setAttribute("aria-label", "Open " + department.name + " department detail");
        path.dataset.index = index;
        path.addEventListener("mouseenter", function (event) {
          state.hoveredName = department.name;
          updateHighlight();
          showTooltip(department, event);
        });
        path.addEventListener("mousemove", function (event) {
          positionTooltip(event);
        });
        path.addEventListener("mouseleave", function () {
          state.hoveredName = null;
          updateHighlight();
          hideTooltip();
        });
        path.addEventListener("focus", function () {
          state.hoveredName = department.name;
          updateHighlight();
          showTooltip(department, null);
        });
        path.addEventListener("blur", function () {
          state.hoveredName = null;
          updateHighlight();
          hideTooltip();
        });
        path.addEventListener("click", function () { openDrawer(department.name); });
        path.addEventListener("keydown", function (event) {
          if (event.key === "Enter" || event.key === " ") {
            event.preventDefault();
            openDrawer(department.name);
          }
        });
        sliceGroup.appendChild(path);
      });
    }

    function drawPie(values) {
      const total = values.reduce(function (sum, value) { return sum + Math.max(0, value); }, 0);
      const outerRadius = 112 + state.reduction * 0.13;
      const innerRadius = 74;
      const gap = 1.6;
      let cursor = 0;
      Array.from(sliceGroup.children).forEach(function (path, index) {
        const value = Math.max(0, values[index] || 0);
        const sweep = total > 0 ? (value / total) * 360 : 0;
        const start = cursor + (sweep > gap ? gap / 2 : 0);
        const end = cursor + sweep - (sweep > gap ? gap / 2 : 0);
        path.setAttribute("d", donutPath(160, 160, outerRadius, innerRadius, start, end));
        cursor += sweep;
      });
    }

    function animatePie(targetValues) {
      if (state.animationFrame) cancelAnimationFrame(state.animationFrame);
      const startValues = state.displayedValues.length
        ? state.displayedValues.slice()
        : targetValues.map(function () { return 0; });
      const startTime = performance.now();
      const duration = window.matchMedia("(prefers-reduced-motion: reduce)").matches ? 1 : 550;

      function frame(now) {
        const raw = Math.min(1, (now - startTime) / duration);
        const eased = 1 - Math.pow(1 - raw, 3);
        state.displayedValues = targetValues.map(function (target, index) {
          return startValues[index] + (target - startValues[index]) * eased;
        });
        drawPie(state.displayedValues);
        if (raw < 1) {
          state.animationFrame = requestAnimationFrame(frame);
        } else {
          state.animationFrame = null;
          state.displayedValues = targetValues.slice();
        }
      }
      state.animationFrame = requestAnimationFrame(frame);
    }

    function priorityPhase() {
      if (state.reduction === 0) return "No additional decrease";
      if (state.reduction <= 25) return "Earlier response: Police protected";
      if (state.reduction <= 50) return "Later response: Police support included";
      return "Structural response: Police included at lower weight";
    }

    function renderCards() {
      cards.innerHTML = "";
      const revenueLoss = PROPERTY_TAX_REVENUE * (state.reduction / 100);
      state.model.forEach(function (department) {
        const button = document.createElement("button");
        button.type = "button";
        button.className = "abi-dept-card" + (state.selectedName === department.name ? " is-selected" : "");

        let badges = "";
        if (department.name === "Building and Fire Safety") {
          badges += '<span class="abi-mini-badge">GF portion only</span>';
        }
        if (state.mode === "service-priority" && department.name === "Police") {
          badges += '<span class="abi-mini-badge is-gold">' +
            (department.modeledReduction === 0 ? "Protected" : "Affected later") + "</span>";
        }
        button.innerHTML =
          '<div class="abi-card-head">' +
            '<span class="abi-dot" style="background:' + department.color + '"></span>' +
            '<span class="abi-card-badges">' + badges + '<span aria-hidden="true">›</span></span>' +

            '<span class="abi-card-share">' + (revenueLoss === 0 ? "0.0" : (department.share * 100).toFixed(1)) + '%</span>' +
            '<span class="abi-card-dollars">' + money.format(department.modeledReduction) + '</span>' +

        button.addEventListener("mouseenter", function () { state.hoveredName = department.name; updateHighlight(); });
        button.addEventListener("mouseleave", function () { state.hoveredName = null; updateHighlight(); });
        button.addEventListener("focus", function () { state.hoveredName = department.name; updateHighlight(); });
        button.addEventListener("blur", function () { state.hoveredName = null; updateHighlight(); });
        button.addEventListener("click", function () { openDrawer(department.name); });
        cards.appendChild(button);
      });
    }

    function updateHighlight() {
      const highlightedName = state.hoveredName || state.selectedName;
      const highlighted = highlightedName
        ? state.model.find(function (department) { return department.name === highlightedName; })
        : null;
      const activeName = highlighted && highlighted.pieValue > 0 ? highlighted.name : null;
      Array.from(sliceGroup.children).forEach(function (path, index) {
        const department = state.model[index];
        path.classList.toggle("is-active", activeName === department.name);
        path.classList.toggle("is-muted", Boolean(activeName && activeName !== department.name));
      });
      root.querySelector("#abi-center-value").textContent =
        compactMoney.format(highlighted ? highlighted.modeledReduction : PROPERTY_TAX_REVENUE * (state.reduction / 100));
      root.querySelector("#abi-center-label").textContent = highlighted
        ? highlighted.modeledReduction === 0 && highlighted.name === "Police" && state.mode === "service-priority"
          ? "Police protected in this phase"
          : highlighted.shortName + " modeled decrease"
        : "Modeled revenue decrease";
    }

    function showTooltip(department, event) {
      const item = state.model.find(function (candidate) { return candidate.name === department.name; });
      if (!item) return;
      root.querySelector("#abi-tooltip-name").textContent = item.name;
      root.querySelector("#abi-tooltip-value").textContent = money.format(item.modeledReduction);
      root.querySelector("#abi-tooltip-share").textContent = item.modeledReduction === 0
        ? "No modeled decrease"
        : (item.share * 100).toFixed(1) + "% of modeled decrease";
      tooltip.classList.add("is-open");
      if (event) {
        positionTooltip(event);
      } else {
        tooltip.style.left = "50%";
        tooltip.style.top = "12px";
        tooltip.style.transform = "translateX(-50%)";
      }
    }

    function positionTooltip(event) {
      if (!tooltip.classList.contains("is-open")) return;
      const rect = chartWrap.getBoundingClientRect();
      const width = tooltip.offsetWidth || 210;
      const height = tooltip.offsetHeight || 82;
      let left = event.clientX - rect.left + 14;
      let top = event.clientY - rect.top - height - 12;
      left = Math.max(8, Math.min(left, rect.width - width - 8));
      top = Math.max(8, Math.min(top, rect.height - height - 8));
      tooltip.style.left = left + "px";
      tooltip.style.top = top + "px";
      tooltip.style.transform = "none";
    }

    function hideTooltip() {
      tooltip.classList.remove("is-open");
    }

    function render() {
      state.model = getModel();

      root.querySelector("#abi-percent").textContent = state.reduction + "%";

      root.querySelector("#abi-total-gap").textContent = money.format(EXISTING_GAP + revenueLoss);

        : "If the decrease follows adopted budget shares";

        ? "This hypothetical scenario protects Police in the earlier response phase, introduces Police support costs later and places sworn operations last. It is not an approved City reduction plan."
        : "This mathematical model distributes the decrease using each department’s adopted share of General Fund operating expenditures. It does not represent the City’s likely reduction order.";

        tab.setAttribute("aria-selected", String(tab.dataset.mode === state.mode));
      });
      animatePie(state.model.map(function (department) { return department.pieValue; }));
      renderCards();
      updateHighlight();
      if (state.selectedName) renderDrawer();
    }

    function scenarioFor(selected) {
      const selectedReduction = selected.modeledReduction;
      const selectedBudgetShare = selectedReduction / selected.budget;
      let level;
      let estimate;

      if (state.mode === "service-priority" && selected.name === "Police") {
        level = selectedReduction === 0 ? "Protected in this phase" : priorityPhase();
      } else if (state.reduction === 0) {
        level = "No additional decrease";
      } else if (selectedBudgetShare <= 0.025) {
        level = "Limited operating response";
      } else if (selectedBudgetShare <= 0.05) {
        level = "Moderate service pressure";
      } else if (selectedBudgetShare <= 0.1) {
        level = "Significant service pressure";
      } else {
        level = "Structural service pressure";
      }

      if (state.reduction === 0) {
        estimate = null;
      } else if (state.mode === "service-priority" && selected.name === "Police") {
        if (selectedReduction === 0) {
          estimate = {
            headline: "No Police reduction is modeled in this earlier response phase.",
            detail: "The service-priority scenario assigns the initial modeled decrease to other General Fund departments. Police support costs enter only after the first threshold."
          };
        } else if (state.reduction <= 50) {
          estimate = {
            headline: "Police support costs enter after the earlier response phase.",
            detail: "Vacancies, civilian non-sworn positions, training, equipment replacement or overtime would be considered before sworn operations. This order is hypothetical and requires City review."
          };
        } else {
          estimate = {
            headline: "Police is included under structural service pressure.",
            detail: "The scenario continues to weight Police below other departments and places sworn operations last. This is a modeling assumption, not an approved reduction plan."
          };
        }
      } else if (selectedBudgetShare <= 0.025) {
        estimate = {
          headline: "Operating controls could be used before public-facing service changes.",
          detail: "Comparable cities have started with vacant positions, purchasing, contracts and other discretionary costs. " + selected.impact.detail
        };
      } else if (selectedBudgetShare <= 0.05) {
        estimate = selected.impact;
      } else if (selectedBudgetShare <= 0.1) {
        estimate = {
          headline: "Several operating and service changes could be combined.",
          detail: selected.impact.headline + " " + selected.impact.detail
        };
      } else {
        estimate = {
          headline: "Structural changes could be required to reach this modeled amount.",
          detail: "Comparable municipal responses at sustained shortfalls have included position reductions, fewer operating days, facility closures and program elimination. " + selected.impact.detail
        };
      }

      return { level: level, estimate: estimate, selectedBudgetShare: selectedBudgetShare };
    }

      state.selectedName = name;
      state.lastFocusedElement = document.activeElement;

      drawer.setAttribute("aria-hidden", "true");

      renderCards();
      updateHighlight();
      if (state.lastFocusedElement && typeof state.lastFocusedElement.focus === "function") {
        state.lastFocusedElement.focus();
      }
    }

          '<div class="abi-examples">' +
            '<a href="https://www.sandiego.gov/mayor/preliminary-budget-fy26-releases" target="_blank" rel="noreferrer">City of San Diego FY 2026 preliminary budget reductions</a>' +
            '<a href="https://services.austintexas.gov/edims/document.cfm?id=125801" target="_blank" rel="noreferrer">City of Austin budget savings actions</a>' +
            '<a href="https://www.seattle.gov/documents/departments/financedepartment/11adoptedbudget/introduction.pdf" target="_blank" rel="noreferrer">City of Seattle adopted budget reductions</a>' +
          '</div>' +

Replace each hypothetical estimate with a City-reviewed impact statement before publication.</p></div>';
    }

    function escapeHTML(value) {
      return String(value)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
    }
